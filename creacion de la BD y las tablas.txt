CREATE DATABASE  colegio;
USE colegio;

SET FOREIGN_KEY_CHECKS = 0;
DROP TABLE IF EXISTS profesores;
DROP TABLE IF EXISTS materias;
SET FOREIGN_KEY_CHECKS = 1;

CREATE TABLE profesores (
    id_profesores      INT AUTO_INCREMENT PRIMARY KEY,
    nombre_del_profesor  VARCHAR(50)  NOT NULL UNIQUE,
    email           VARCHAR(100) NOT NULL UNIQUE,
    grado_y_seccion_acargo           INT NOT NULL DEFAULT 1
);

CREATE TABLE materias (
    id_materia        INT AUTO_INCREMENT PRIMARY KEY,
    nombre_materia          VARCHAR(80) NOT NULL
);

CREATE TABLE Items (
    item_id         INT AUTO_INCREMENT PRIMARY KEY,
    nombre          VARCHAR(80) NOT NULL,
    tipo            VARCHAR(30) NOT NULL,
    rareza          VARCHAR(20) NOT NULL DEFAULT 'comun',
    precio_monedas  INT NOT NULL DEFAULT 0
